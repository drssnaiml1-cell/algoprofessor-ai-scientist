\begin{frame}{What is machine learning?}
\begin{bdef}[Machine learning]
A program \emph{learns} when its performance on a task, measured by some score, improves with experience (data). Instead of writing the rules, we give examples and an algorithm finds the rules.
\end{bdef}
\begin{columns}[T]
\column{0.32\textwidth}
\begin{bex}[Supervised]
Examples come with answers (labels).\\ \emph{Spam / not spam; house price.}
\end{bex}
\column{0.32\textwidth}
\begin{bex}[Unsupervised]
No labels; find structure.\\ \emph{Customer segments; anomalies.}
\end{bex}
\column{0.32\textwidth}
\begin{bex}[Reinforcement]
Learn by trial and reward.\\ \emph{Game play; robot control; ad bidding.}
\end{bex}
\end{columns}
\begin{bmath}[Supervised learning in one line]
Given $\mathcal{D}=\{(x_i,y_i)\}_{i=1}^{n}$, find $f_\theta$ that minimises the average loss $\;\frac1n\sum_i \ell\big(f_\theta(x_i),y_i\big)$.
\end{bmath}
\end{frame}

\begin{frame}{Kinds of data}
\begin{columns}[T]
\column{0.5\textwidth}
\begin{bdef}[Data types]
\textbf{Numerical} (age, income) $\cdot$ \textbf{categorical} (city, contract) $\cdot$ \textbf{ordinal} (low $<$ medium $<$ high) $\cdot$ \textbf{text} $\cdot$ \textbf{image} $\cdot$ \textbf{time series}
\end{bdef}
\begin{bex}[One Adult Income row]
age 39 $\cdot$ workclass \emph{State-gov} $\cdot$ education \emph{Bachelors} $\cdot$ 40 hours/week $\cdot$ income \emph{$\le$50K}
\end{bex}
\column{0.46\textwidth}
\begin{bdef}[Tabular data]
Rows = examples (people, houses, customers). Columns = features. One column = the target $y$ we want to predict.
\end{bdef}
\begin{bwarn}[Feature or target?]
A column known only \emph{after} the event (e.g.\ ``refund issued'' when predicting churn) leaks the answer.
\end{bwarn}
\end{columns}
\end{frame}

\begin{frame}{Descriptive statistics --- the mathematics of EDA}
\begin{bmath}[Centre, spread and relationship]
Mean $\bar x=\frac1n\sum_i x_i$ \qquad Variance $s^2=\frac{1}{n-1}\sum_i (x_i-\bar x)^2$ \qquad Standard deviation $s=\sqrt{s^2}$\\[3pt]
Median = middle value; quantile $q_p$ = value below which a share $p$ of the data lies (robust to outliers)\\[3pt]
Pearson correlation $\;r=\dfrac{\sum_i (x_i-\bar x)(y_i-\bar y)}{\sqrt{\sum_i (x_i-\bar x)^2}\sqrt{\sum_i (y_i-\bar y)^2}}\in[-1,1]$
\end{bmath}
\begin{bex}[Worked example]
Ages $3,4,5,6,32$: mean $=10$, median $=5$. One outlier moves the mean a lot and the median not at all --- so report both.
\end{bex}
\end{frame}

\begin{frame}[fragile]{Vectorisation with NumPy}
\begin{columns}[T]
\column{0.48\textwidth}
\begin{bdef}[Vectorisation]
Apply one operation to a whole array in compiled code instead of looping in Python.
\end{bdef}
\begin{bmath}[Dot product]
$x\cdot x=\sum_{i=1}^{n}x_i^2$ --- one call, $n$ multiplications.
\end{bmath}
\begin{bres}
Loop 135.2 ms $\rightarrow$ NumPy 0.9 ms: \textbf{154$\times$ faster}, same answer.
\end{bres}
\column{0.48\textwidth}
\begin{bcode}
\begin{lstlisting}
import numpy as np
x = np.random.rand(1_000_000)
s_loop = sum(v * v for v in x)   # slow
s_vec  = np.dot(x, x)            # fast
z = (x - x.mean()) / x.std()     # no loop needed
\end{lstlisting}
\end{bcode}
\end{columns}
\end{frame}

\begin{frame}{The EDA algorithm}
\begin{columns}[T]
\column{0.52\textwidth}

\begin{algorithmic}[1]
\Require table $T$, target column $y$
\State Profile shape, types, unique values
\State Count missing values; ask \emph{why}
\State Remove exact duplicate rows
\State Check target balance
\For{each feature $x_j$}
  \State Plot its distribution; note outliers
  \State Plot $y$ against $x_j$
\EndFor
\State Write 3 findings and 3 questions
\end{algorithmic}
\end{balgo}
\column{0.44\textwidth}

\textbf{MCAR}: completely at random. \textbf{MAR}: depends on other columns. \textbf{MNAR}: depends on the missing value itself (e.g.\ high earners skip income).
\end{bdef}

Fill missing \texttt{occupation} with \emph{Unknown}: keeps rows and lets the model learn from missingness.

\begin{frame}{Linear regression}
\begin{bdef}[Regression]
Predict a \emph{number}. Linear regression assumes the target is a weighted sum of the features plus noise.
\end{bdef}
\begin{bmath}[Model, loss and closed-form solution]
$\hat y=\theta_0+\theta_1x_1+\dots+\theta_dx_d=x^\top\theta$ \qquad
$\mathrm{MSE}(\theta)=\frac1n\sum_i(y_i-x_i^\top\theta)^2=\frac1n\lVert y-X\theta\rVert^2$\\[3pt]
Setting the gradient to zero gives the normal equation $\;\theta^\ast=(X^\top X)^{-1}X^\top y$.
\end{bmath}
\begin{bex}[Intuition]
Price $\approx \theta_0 + \theta_1\cdot\text{median income} + \theta_2\cdot\text{rooms per household}+\dots$ --- each $\theta_j$ says how much the price moves per unit of that feature, holding the others fixed.
\end{bex}
\end{frame}

\begin{frame}{Gradient descent --- how most models learn}
\begin{columns}[T]
\column{0.5\textwidth}
\begin{bmath}[Update rule]
$\nabla_\theta\,\mathrm{MSE}=-\frac{2}{n}X^\top(y-X\theta)$\\[3pt]
$\theta\leftarrow\theta-\eta\,\nabla_\theta\,\mathrm{MSE}$, \;learning rate $\eta>0$
\end{bmath}
\begin{bex}[One step by hand]
$f(\theta)=(\theta-3)^2$, $\theta=0$, $\eta=0.1$: gradient $=-6$, new $\theta=0.6$. Repeat $\rightarrow 3$.
\end{bex}
\column{0.46\textwidth}
\begin{balgo}[Batch gradient descent]
\begin{algorithmic}[1]
\Require $X,y$, rate $\eta$, steps $T$
\State $\theta\gets 0$
\For{$t=1$ to $T$}
  \State $g\gets -\frac{2}{n}X^\top(y-X\theta)$
  \State $\theta\gets\theta-\eta g$
\EndFor
\State \Return $\theta$
\end{algorithmic}
\end{balgo}
\end{columns}
\end{frame}

\begin{frame}{Generalisation: bias, variance and cross-validation}
\begin{columns}[T]
\column{0.5\textwidth}
\begin{bdef}[Overfitting]
A model that fits training data well but new data badly has memorised noise.
\end{bdef}
\begin{bmath}[Bias--variance decomposition]
$\mathbb{E}\big[(y-\hat f(x))^2\big]=\underbrace{\mathrm{Bias}^2}_{\text{too simple}}+\underbrace{\mathrm{Var}}_{\text{too flexible}}+\sigma^2_{\text{noise}}$
\end{bmath}
\column{0.46\textwidth}
\begin{balgo}[$k$-fold cross-validation]
\begin{algorithmic}[1]
\State Split training data into $k$ folds
\For{$i=1$ to $k$}
  \State Train on all folds except $i$
  \State Score on fold $i$
\EndFor
\State \Return mean and std of the $k$ scores
\end{algorithmic}
\end{balgo}
\begin{bwarn}
Use the test set \textbf{once}, at the end.
\end{bwarn}
\end{columns}
\end{frame}

\begin{frame}{Trees, forests and gradient boosting}
\begin{columns}[T]
\column{0.5\textwidth}
\begin{bdef}[Decision tree]
Splits data with yes/no questions (``income $>$ 5?'') chosen to make each branch as pure as possible.
\end{bdef}
\begin{bdef}[Ensembles]
\textbf{Random forest}: many trees on bootstrap samples, averaged (reduces variance).
\textbf{Boosting}: trees added one by one, each fixing the previous errors (reduces bias).
\end{bdef}
\begin{bmath}[XGBoost objective]
$\sum_i \ell(y_i,\hat y_i)+\sum_k\big(\gamma T_k+\tfrac12\lambda\lVert w_k\rVert^2\big)$\\ ($T_k$ leaves, $w_k$ leaf weights of tree $k$)
\end{bmath}
\column{0.46\textwidth}
\begin{balgo}[Gradient boosting (squared loss)]
\begin{algorithmic}[1]
\State $F_0(x)\gets \bar y$
\For{$m=1$ to $M$}
  \State residuals $r_i\gets y_i-F_{m-1}(x_i)$
  \State fit a small tree $h_m$ to $(x_i,r_i)$
  \State $F_m\gets F_{m-1}+\eta\,h_m$
\EndFor
\State \Return $F_M$
\end{algorithmic}
\end{balgo}
\end{columns}
\end{frame}

\begin{frame}{Logistic regression}
\begin{bdef}[Classification]
Predict a \emph{category}. Binary classification outputs the probability of the positive class (here: the customer churns).
\end{bdef}
\begin{columns}[T]
\column{0.55\textwidth}
\begin{bmath}[Sigmoid and log loss]
$p=\sigma(z)=\dfrac{1}{1+e^{-z}},\quad z=x^\top\theta$\\[4pt]
$\mathcal{L}=-\frac1n\sum_i\big[y_i\log p_i+(1-y_i)\log(1-p_i)\big]$\\[3pt]
Predict ``churn'' when $p\ge t$ (threshold, default 0.5).
\end{bmath}
\column{0.41\textwidth}
\begin{bex}[Reading a coefficient]
$\theta_j=0.7$ for ``month-to-month contract'' multiplies the odds of churning by $e^{0.7}\approx2.0$.
\end{bex}
\end{columns}
\end{frame}

\begin{frame}{The confusion matrix and the metrics built on it}
\begin{columns}[T]
\column{0.42\textwidth}
\centering
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{c|c|c}
 & \textbf{pred.\ churn} & \textbf{pred.\ stay}\\\hline
\textbf{churned} & TP & FN\\\hline
\textbf{stayed} & FP & TN\\
\end{tabular}
\begin{bex}[Worked example]
100 churners: model catches 60 (TP), misses 40 (FN), and wrongly flags 20 stayers (FP).\\
$P=60/80=0.75$, \;$R=60/100=0.60$, \;$F_1=0.667$.
\end{bex}
\column{0.54\textwidth}
\begin{bmath}[Metrics]
Precision $P=\frac{TP}{TP+FP}$ \quad Recall $R=\frac{TP}{TP+FN}$\\[3pt]
$F_1=\frac{2PR}{P+R}$ \quad Accuracy $=\frac{TP+TN}{\text{all}}$\\[3pt]
ROC-AUC $=P(\text{score of a random churner}>\text{score of a random stayer})$
\end{bmath}
\begin{bwarn}[Accuracy hides failure]
A model that never predicts churn is 73\% accurate --- and useless.
\end{bwarn}
\end{columns}
\end{frame}

\hourslide{4}{BUILD}{Imbalance, hyperparameter search, explainability}{IBM Telco Customer Churn}{Class weighting, Optuna, SHAP}

\begin{frame}{Class imbalance}
\begin{bdef}[Imbalanced data]
One class is much rarer than the other (here 26.5\% churners). Models drift towards predicting the majority.
\end{bdef}
\begin{columns}[T]
\column{0.5\textwidth}
\begin{bmath}[Weighted loss]
$\mathcal{L}_w=-\frac1n\sum_i\big[w_1\,y_i\log p_i+(1-y_i)\log(1-p_i)\big]$\\[3pt]
with $w_1=\frac{\#\text{negatives}}{\#\text{positives}}=2.77$ here.
\end{bmath}
\column{0.46\textwidth}
\begin{bex}[Three remedies]
\textbf{Re-weight} the loss $\cdot$ \textbf{re-sample} (SMOTE, undersampling) $\cdot$ \textbf{move the threshold} $t$ to suit the cost of errors.
\end{bex}
\end{columns}
\end{frame}

\begin{frame}[fragile]{Hyperparameter optimisation}
\begin{columns}[T]
\column{0.48\textwidth}
\begin{bdef}[Hyperparameter]
A setting chosen \emph{before} training (tree depth, learning rate), not learned from the data.
\end{bdef}
\begin{balgo}[Bayesian optimisation (Optuna TPE)]
\begin{algorithmic}[1]
\For{trial $=1$ to $N$}
  \State propose $\lambda$ likely to score well
  \State score $\lambda$ by cross-validation
  \State update model of good vs bad regions
\EndFor
\State \Return best $\lambda$
\end{algorithmic}
\end{balgo}
\column{0.48\textwidth}
\begin{bcode}
\begin{lstlisting}
def objective(trial):
    params = dict(
      max_depth=trial.suggest_int("max_depth", 2, 6),
      learning_rate=trial.suggest_float(
          "learning_rate", 0.01, 0.2, log=True))
    return cross_val_score(make_pipe(params),
        X_tr, y_tr, cv=4, scoring="roc_auc").mean()
study = optuna.create_study(direction="maximize")
study.optimize(objective, n_trials=25)
\end{lstlisting}
\end{bcode}
\end{columns}
\end{frame}

\begin{frame}{Explainability with SHAP}
\begin{bdef}[SHAP value]
How much feature $j$ pushed this prediction up or down compared with the average prediction --- based on Shapley values from game theory.
\end{bdef}
\begin{columns}[T]
\column{0.56\textwidth}
\begin{bmath}[Shapley value and additivity]
$\phi_j=\sum_{S\subseteq F\setminus\{j\}}\frac{|S|!\,(|F|-|S|-1)!}{|F|!}\big[f(S\cup\{j\})-f(S)\big]$\\[3pt]
$f(x)=\phi_0+\sum_j\phi_j(x)$
\end{bmath}
\column{0.4\textwidth}
\begin{bex}[Two features]
Average churn score 0.27. For one customer: contract $+0.30$, tenure $+0.12$ $\Rightarrow$ $0.27+0.30+0.12=0.69$.
\end{bex}
\end{columns}
\end{frame}

\begin{frame}{Formula sheet --- Sunday 1}
\renewcommand{\arraystretch}{1.35}
\begin{tabularx}{\linewidth}{l X}
\toprule
Mean / variance & $\bar x=\frac1n\sum x_i$, \; $s^2=\frac1{n-1}\sum(x_i-\bar x)^2$\\
Linear model / MSE & $\hat y=x^\top\theta$, \; $\frac1n\lVert y-X\theta\rVert^2$, \; $\theta^\ast=(X^\top X)^{-1}X^\top y$\\

Regression & RMSE, MAE, \; $R^2=1-\frac{SS_{res}}{SS_{tot}}$\\
SHAP & $f(x)=\phi_0+\sum_j\phi_j(x)$\\

\end{tabularx}
