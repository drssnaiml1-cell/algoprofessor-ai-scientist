\hourslide{3}{BUILD}{Churn project: problem framing \& baseline}{IBM Telco Customer Churn (7,043 customers)}{Problem framing, baseline thinking, F1 / ROC-AUC}

\begin{frame}{Frame it before you model it}
\begin{columns}[T]
\column{0.5\textwidth}
\begin{block}{The question}
Which customers will cancel next month, so retention can call them first?
\end{block}
\textbf{Metrics:} ROC-AUC (ranking) and F1 for churners.\\[4pt]
$F_1=\dfrac{2PR}{P+R}$
\column{0.46\textwidth}
\begin{exampleblock}{Our run}
Churn rate 26.5\%\\
Month-to-month 42.7\% $\cdot$ two-year 2.8\%\\[3pt]
Dummy: AUC 0.500, F1 0.000\\
Logistic: AUC 0.846, F1 0.607
\end{exampleblock}
\begin{alertblock}{Accuracy hides failure}
Never predicting churn is 73\% ``accurate''.
\end{alertblock}
\end{columns}
\end{frame}

\begin{block}{Lesson}
