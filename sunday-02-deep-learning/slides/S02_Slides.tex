\begin{frame}[fragile]{Five lines run every neural network}
\begin{columns}[T]
\column{0.5\textwidth}
\begin{lstlisting}
for xb, yb in batches:
    out  = model(xb)              # forward
    loss = F.cross_entropy(out, yb)
    opt.zero_grad()
    loss.backward()               # autograd
    opt.step()                    # update
\end{lstlisting}
Cross-entropy: $L=-\log p_y$
\column{0.46\textwidth}
\begin{exampleblock}{Our run}

MLP 784--256--128--10:\\ \textbf{87.1\% test accuracy} in 5 epochs
\end{exampleblock}
\end{columns}

\hourslide{7}{TEACH}{CNNs, augmentation and mixed precision}{CIFAR-10 (10,000-image split)}{CNN design, augmentation, AMP training}

\begin{frame}{Why convolutions work on images}
\begin{columns}[T]
\column{0.5\textwidth}
\begin{itemize}
\item \textbf{Weight sharing:} one filter everywhere
\item \textbf{Locality:} each filter sees a small patch
\item Conv $\rightarrow$ BatchNorm $\rightarrow$ ReLU $\rightarrow$ Pool
\item Edges $\rightarrow$ textures $\rightarrow$ parts $\rightarrow$ objects
\end{itemize}
\begin{block}{Augmentation}
Random flips + shifted crops: learn the object, not the pixels.
\end{block}
\column{0.46\textwidth}
\begin{exampleblock}{Our run}
289,194-parameter CNN, 8 epochs on 8,000 images:\\ \textbf{$\approx$67\% held-out accuracy}\\[3pt]
bfloat16 vs fp32: same prediction on \textbf{99.0\%} of images
\end{exampleblock}
\begin{alertblock}{Leakage}
Normalise with \emph{training} mean and std only.
\end{alertblock}
\end{columns}
\end{frame}

\item De-duplicate \emph{before} splitting
\item Stratified 70 / 15 / 15

\begin{frame}{Lab accuracy is not field accuracy}
\begin{columns}[T]
\column{0.5\textwidth}

\begin{tabular}{lrr}

Condition & CNN & ViT\\
\midrule

\begin{alertblock}{The day's most important result}
The ``best'' model collapses to 20\% with mild noise.
\end{alertblock}
\begin{block}{Fixes}
Noise/brightness/blur augmentation $\cdot$ real field photos $\cdot$ a ``not sure'' answer
\end{block}
