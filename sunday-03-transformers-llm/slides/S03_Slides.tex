\begin{frame}{What is a language model?}
\begin{bdef}[Language model]
A model that gives the probability of the next token given all previous tokens. Generating text = repeatedly sampling the next token.
\end{bdef}
\begin{columns}[T]
\column{0.55\textwidth}
\begin{bmath}[Chain rule of probability and perplexity]
$P(x_1,\dots,x_T)=\prod_{t=1}^{T}P(x_t\mid x_{<t})$\\[3pt]
Training loss $\mathcal{L}=-\frac1T\sum_t\log P(x_t\mid x_{<t})$\\[3pt]
Perplexity $=e^{\mathcal{L}}$ (``how many choices the model is torn between'')
\end{bmath}
\column{0.41\textwidth}
\begin{bex}[Example]
After ``ROMEO:'' and a line break, a good model puts high probability on capital letters. Uniform guessing over 65 characters has perplexity \textbf{65}.
\end{bex}
\end{columns}
\end{frame}

\begin{frame}{Tokenisation and embeddings}
\begin{columns}[T]
\column{0.5\textwidth}
\begin{bdef}[Token]
The unit a model reads: a character, a word, or a sub-word piece.
\end{bdef}
\begin{balgo}[Byte-pair encoding (BPE)]
\begin{algorithmic}[1]
\State start with characters + end-of-word mark
\For{$k=1$ to $K$ merges}
  \State count adjacent symbol pairs
  \State merge the most frequent pair
\EndFor
\State \Return merge list (the vocabulary)
\end{algorithmic}
\end{balgo}
\column{0.46\textwidth}
\begin{bres}
First merges on Shakespeare: \texttt{e</w>}, \texttt{s</w>}, \texttt{th}, \texttt{t</w>}, \texttt{d</w>}, \ldots, \texttt{the</w>}.
\end{bres}
\begin{bdef}[Embedding]
A learned vector per token: $e_i=E[i]$, $E\in\mathbb{R}^{V\times d}$. Similar tokens end up with similar vectors.
\end{bdef}
\end{columns}
\end{frame}

\begin{frame}{Scaled dot-product attention}
\begin{bdef}[Attention]
Each token asks a question (query), every token offers a label (key) and a message (value). A token collects the messages whose keys match its query.
\end{bdef}
\begin{columns}[T]
\column{0.55\textwidth}
\begin{bmath}[Formula]
$Q=XW_Q,\;K=XW_K,\;V=XW_V$\\[2pt]
$\mathrm{Attention}=\mathrm{softmax}\!\Big(\frac{QK^\top}{\sqrt{d_k}}+M\Big)V$\\[2pt]
$M_{ij}=-\infty$ for $j>i$ (causal mask)
\end{bmath}
\column{0.41\textwidth}
\begin{bex}[Two tokens by hand]
$q_2\cdot k_1=1,\;q_2\cdot k_2=3$, $d_k=4$: scores $0.5, 1.5$; softmax $\rightarrow 0.27, 0.73$. Token 2 takes 73\% of its own value and 27\% of token 1's.
\end{bex}
\end{columns}
\end{frame}

\begin{frame}{The Transformer block and positions}
\begin{columns}[T]
\column{0.5\textwidth}
\begin{balgo}[GPT forward pass]
\begin{algorithmic}[1]
\State $h\gets$ token embedding $+$ position embedding
\For{each of $L$ blocks}
  \State $h\gets h+\mathrm{MultiHeadAttn}(\mathrm{LN}(h))$
  \State $h\gets h+\mathrm{MLP}(\mathrm{LN}(h))$
\EndFor
\State logits $\gets\mathrm{LN}(h)\,W_{out}$
\end{algorithmic}
\end{balgo}
\begin{bdef}[Multi-head]
Several attentions in parallel, each free to focus on a different pattern (rhyme, speaker, grammar).
\end{bdef}
\column{0.46\textwidth}
\begin{bmath}[Sinusoidal positions (original Transformer)]
$PE_{(pos,2i)}=\sin\!\big(pos/10000^{2i/d}\big)$\\
$PE_{(pos,2i+1)}=\cos\!\big(pos/10000^{2i/d}\big)$
\end{bmath}
\small Our MiniGPT learns its position embeddings instead.
\begin{bdef}[Residual + LayerNorm]
Residual paths let gradients flow through deep stacks; LayerNorm keeps activations at a stable scale.
\end{bdef}
\end{columns}
\end{frame}

\begin{frame}{Encoders, decoders and transfer learning}
\begin{columns}[T]
\column{0.5\textwidth}
\begin{bdef}[Encoder vs decoder]
\textbf{Encoder} (BERT-style): reads the whole text both ways --- best for classification and search. \textbf{Decoder} (GPT-style): reads left to right --- best for generation.
\end{bdef}
\begin{bdef}[Transfer learning]
Pretrain on huge unlabelled text, then fine-tune on a small labelled task.
\end{bdef}
\column{0.46\textwidth}
\begin{bmath}[TF-IDF baseline]
$\mathrm{tfidf}(t,d)=\mathrm{tf}(t,d)\cdot\log\dfrac{N}{\mathrm{df}(t)}$\\[2pt]
Words frequent in a review but rare overall get high weight.
\end{bmath}
\begin{bex}[Classify from \texttt{[CLS]}]
Prepend \texttt{[CLS]}, run the encoder, feed the \texttt{[CLS]} vector to a linear layer $\rightarrow$ positive / negative.
\end{bex}
\end{columns}
\end{frame}

\begin{frame}[fragile]{Code, result and products}
\begin{columns}[T]
\column{0.52\textwidth}
\begin{bcode}
\begin{lstlisting}
tfidf = TfidfVectorizer(ngram_range=(1, 2), min_df=2)
lr = LogisticRegression(C=4).fit(tfidf.fit_transform(X), y)
# in class, a pretrained encoder:

\end{lstlisting}

A transformer trained from scratch on 7,000 sentences loses to TF-IDF. Transformers win through \textbf{pretraining}; a fine-tuned DistilBERT typically reaches about 0.91.
\end{bwarn}
\begin{bapp}
Review and social-media sentiment monitoring; support-ticket routing; spam and intent detection; Hugging Face hosts thousands of pretrained encoders.
\end{bapp}
\end{columns}
\end{frame}

\begin{frame}{Instruction tuning and LoRA}
\begin{bdef}[Instruction tuning]
Fine-tune a pretrained model on (instruction, response) pairs so it follows a task format instead of just continuing text.
\end{bdef}
\begin{columns}[T]
\column{0.55\textwidth}
\begin{bmath}[LoRA: low-rank adaptation]
$h=Wx+\frac{\alpha}{r}BAx$, \; $W$ frozen,\\ $A\in\mathbb{R}^{r\times k}$, $B\in\mathbb{R}^{d\times r}$, $B_0=0$\\[3pt]
Trainable per layer: $r(d+k)$ instead of $dk$
\end{bmath}
\column{0.41\textwidth}
\begin{bex}[Worked example]
$d=k=4096$, $r=8$: full $=16.8$\,M weights; LoRA $=8\cdot8192=65{,}536$ (\textbf{0.4\%}).
\end{bex}
\end{columns}
\begin{bdef}[QLoRA]
LoRA on top of a base model stored in 4-bit --- fits multi-billion-parameter models on one GPU.
\end{bdef}
\end{frame}

\begin{frame}[fragile]{Training algorithm and code}
\begin{columns}[T]
\column{0.46\textwidth}
\begin{balgo}[LoRA instruction tuning]
\begin{algorithmic}[1]
\State freeze all base weights
\State wrap attention / MLP layers with $A,B$
\For{each batch of templated examples}
  \State targets of prompt tokens $\gets -100$
  \State loss on answer tokens only
  \State update $A,B$ only
\EndFor
\end{algorithmic}
\end{balgo}
\begin{bcode}[Chat template]
\begin{lstlisting}
### Review:
a stirring , funny and finally ...
### Sentiment: positive
\end{lstlisting}
\end{bcode}
\column{0.5\textwidth}
\begin{bcode}
\begin{lstlisting}
class LoRALinear(nn.Module):
    def __init__(self, base, r=8, alpha=16):
        super().__init__(); self.base = base
        self.scale = alpha / r
        self.A = nn.Parameter(torch.randn(r, base.in_features)*0.01)
        self.B = nn.Parameter(torch.zeros(base.out_features, r))
    def forward(self, x):
        return self.base(x) + (x @ self.A.T @ self.B.T) * self.scale
\end{lstlisting}
\end{bcode}
\begin{bres}
Trainable: \rLoraParams{} of 891,969 (\rLoraPct\%).
\end{bres}
\end{columns}
\end{frame}

\begin{frame}{Memory, pitfalls and products}
\begin{columns}[T]
\column{0.5\textwidth}
\begin{bmath}[GPU memory rule of thumb (AdamW)]
Full fine-tune $\approx$ 16 bytes/parameter\\ (weights, gradients, two Adam moments)\\[2pt]
0.5\,B model: $\approx$ 8\,GB full vs $\approx$ 1\,GB fp16 weights + small adapters with LoRA
\end{bmath}
\begin{bapp}
Domain assistants (legal, medical, customer support) built by adapting open models; tools: Hugging Face PEFT, Unsloth, Axolotl.
\end{bapp}
\column{0.46\textwidth}
\begin{bwarn}[Our first attempt failed]
Loss on the whole text $\Rightarrow$ 48\% accuracy, every review ``negative''. The learning signal went into predicting review text. Masking the prompt fixed it --- this is why libraries mask prompts.
\end{bwarn}
\end{columns}
\end{frame}

\begin{frame}{How to score an LLM}
\begin{columns}[T]
\column{0.5\textwidth}
\begin{bdef}[Likelihood scoring]
For a multiple-choice task, pick the answer the model finds most probable after the prompt.
\end{bdef}
\begin{bmath}[Score]
$\hat a=\arg\max_{a}\sum_{t\in a}\log P(t\mid\text{prompt},\,a_{<t})$
\end{bmath}
\begin{bdef}[Contamination]
Test items (or near-copies) present in training data. Scores then measure memory, not skill.
\end{bdef}
\column{0.46\textwidth}
\begin{balgo}[Evaluation harness]
\begin{algorithmic}[1]
\State freeze test set and prompts
\State run every system, same settings
\State add simple baselines
\State check train--test overlap
\State check forgetting on the old task
\State repeat with several seeds
\end{algorithmic}
\end{balgo}
\end{columns}
\end{frame}

\begin{frame}{Decoding strategies}
\begin{columns}[T]
\column{0.5\textwidth}
\begin{bdef}[Decoding]
How the next token is chosen from the model's probabilities: greedy, sampling, top-$k$, top-$p$.
\end{bdef}
\begin{bmath}[Temperature, top-$k$, top-$p$]
$p_i\propto e^{z_i/T}$ \; (low $T$: safe; high $T$: creative)\\
top-$k$: keep the $k$ most likely tokens\\
top-$p$: keep the smallest set with total probability $\ge p$
\end{bmath}
\column{0.46\textwidth}
\begin{bex}[Logits $(2,1,0)$]
$T=1$: $(0.67, 0.24, 0.09)$\\ $T=0.5$: $(0.87, 0.12, 0.02)$ --- sharper.
\end{bex}
\begin{bres}
$T=0.8$ sample after ``KING:'' --- ``'Tis not to tell the Lord the night way my hand''.
\end{bres}
\end{columns}
\end{frame}

\begin{frame}{Check yourself}
\begin{enumerate}
\item A model has validation loss 1.642 nats per character. What is its perplexity?
\item Why divide attention scores by $\sqrt{d_k}$?
\item How many trainable parameters does LoRA with $r=16$ add to a $1024\times1024$ layer?
\item Why must the prompt tokens be masked during instruction tuning?
\item What does a 0.3\% train--test overlap tell you about our SST-2 results?
\end{enumerate}
\begin{bex}[Answers]
1. $e^{1.642}\approx5.2$. \;2. Large dot products saturate the softmax and kill gradients. \;3. $16\times2048=32{,}768$. \;4. Otherwise the loss is dominated by predicting the prompt, not the answer. \;5. Contamination is negligible; the scores are not inflated by memory.
\end{bex}
\end{frame}
