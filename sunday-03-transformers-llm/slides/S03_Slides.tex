\hourslide{11}{TEACH}{Self-attention, tokenisation and a mini-GPT}{Tiny Shakespeare (1.1 M characters)}{Attention maths, tokenisers, GPT internals}

\begin{frame}{Attention in one equation}
\[
\mathrm{Attention}(Q,K,V)=\mathrm{softmax}\!\Big(\frac{QK^{\top}}{\sqrt{d_k}}+M\Big)V
\]
\begin{columns}[T]
\column{0.48\textwidth}
\begin{itemize}
\item $Q$: what I am looking for
\item $K$: what I contain
\item $V$: what I pass on
\item $M$: causal mask, $-\infty$ above the diagonal
\end{itemize}
\column{0.48\textwidth}
\begin{block}{BPE tokenisation}
Merge the most frequent pair, repeat. First merges on Shakespeare: \texttt{e</w>}, \texttt{s</w>}, \texttt{th}, \ldots, \texttt{the</w>}
\end{block}
\end{columns}
\end{frame}
