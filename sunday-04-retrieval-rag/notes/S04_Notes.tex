\hourhead{16}{TEACH}{Embeddings and vector search}{20,000 synthetic 64-d vectors}{Vector maths, FAISS, ANN trade-offs}

\topic{Embeddings and similarity}
\begin{bdef}[Text embedding]
A fixed-length vector representing the meaning of a piece of text. Texts with similar meaning map to nearby vectors, even if they share no words.
\end{bdef}
Sunday 3's token embeddings described single tokens. A sentence embedding model pools a transformer's outputs into one vector per passage and is trained (often contrastively) so that a question and its answer land close together. That turns ``find relevant text'' into ``find nearest vectors''.
\begin{bmath}[Similarity measures]
Cosine: $\cos(u,v)=\dfrac{u\cdot v}{\lVert u\rVert\,\lVert v\rVert}$ \qquad Euclidean: $\lVert u-v\rVert_2$\\[3pt]
For unit vectors, $\lVert u-v\rVert^2=2-2\cos(u,v)$, so ranking by either gives the same order.
\end{bmath}
\begin{bex}[By hand]
$u=(1,0)$, $v=(0.6,0.8)$: $\cos=0.6$. $w=(-1,0)$: $\cos(u,w)=-1$ (opposite meaning direction).
\end{bex}

\topic{Exact and approximate nearest-neighbour search}
\begin{bdef}[ANN index]
A data structure that finds \emph{almost} the nearest vectors much faster than comparing with every one. \textbf{IVF} clusters vectors and searches only a few clusters; \textbf{HNSW} walks a multi-layer graph of neighbours.
\end{bdef}
\begin{bmath}[Recall@$k$ for an index]
$\text{Recall@}k=\dfrac{|\text{ANN top-}k\cap\text{exact top-}k|}{k}$ --- what fraction of the true neighbours the fast index found.
\end{bmath}
\begin{balgo}[IVF search]
\begin{algorithmic}[1]
\State \textbf{build:} $k$-means on the vectors $\rightarrow$ $n_{list}$ centroids; store each vector in its centroid's list
\State \textbf{query:} find the $n_{probe}$ nearest centroids
\State compare the query only with vectors in those lists; return the top $k$
\end{algorithmic}
\end{balgo}
\begin{bcode}
\begin{lstlisting}
flat = faiss.IndexFlatIP(d); flat.add(X)                           # exact
ivf  = faiss.IndexIVFFlat(faiss.IndexFlatIP(d), d, 100); ivf.train(X); ivf.add(X); ivf.nprobe = 8
hnsw = faiss.IndexHNSWFlat(d, 32); hnsw.add(X)
D, I = hnsw.search(Q, 10)
\end{lstlisting}
\end{bcode}
\begin{bres}[20,000 vectors, 200 queries]
\begin{tabular}{lrr}
\toprule
Index & Recall@10 & $\mu$s per query\\
\midrule
Flat (exact) & 1.000 & 465.6\\
IVF ($n_{probe}=8$) & 0.842 & 47.0\\
HNSW ($M=32$) & \textbf{0.980} & \textbf{43.6}\\
\bottomrule
\end{tabular}\\[3pt]
HNSW is about $10\times$ faster than exact search while losing only 2\% of true neighbours.
\end{bres}
\begin{bapp}
Vector databases --- FAISS, pgvector, Pinecone, Weaviate, Milvus, Qdrant --- power semantic search, recommendations and RAG in production.
\end{bapp}

\hourhead{17}{TEACH}{Keyword, dense and hybrid retrieval}{TruthfulQA --- 790 questions, 4,000+ answers}{BM25, dense retrieval, fusion, IR metrics}

\topic{BM25: strong keyword search}
\begin{bdef}[BM25]
A keyword ranking function: a document scores high if it contains the query words often, if those words are rare overall, and if the document is not just long.
\end{bdef}
\begin{bmath}[BM25 score]
$\mathrm{score}(q,d)=\sum_{t\in q}\mathrm{IDF}(t)\cdot\dfrac{f(t,d)\,(k_1+1)}{f(t,d)+k_1\big(1-b+b\,\frac{|d|}{\mathrm{avgdl}}\big)}$, \quad typically $k_1=1.5$, $b=0.75$.
\end{bmath}
The fraction saturates: the tenth occurrence of a word adds far less than the first. The $b$ term penalises long documents, which match more words simply by being long.

\topic{Hybrid retrieval and metrics}
\begin{bmath}[Reciprocal rank fusion (RRF)]
$\mathrm{RRF}(d)=\sum_{r\in\{\text{BM25},\text{dense}\}}\dfrac{1}{60+\mathrm{rank}_r(d)}$ --- combine rankings without having to calibrate their scores.
\end{bmath}
\begin{bmath}[Retrieval metrics]
Recall@$k$: is a relevant document in the top $k$? \quad MRR $=\frac1{|Q|}\sum_q\frac{1}{\mathrm{rank}_q}$\\[2pt]
nDCG@$k$ $=\dfrac{\sum_{i\le k}rel_i/\log_2(i+1)}{\text{ideal DCG}}$ --- rewards relevant results near the top.
\end{bmath}
\begin{bex}[MRR by hand]
First relevant answers at ranks 1, 3 and 2 for three queries: MRR $=(1+\tfrac13+\tfrac12)/3=0.61$.
\end{bex}
\begin{bcode}
\begin{lstlisting}
bm25 = BM25Okapi([tokenize(a) for a in answers])
dense_scores = emb(answers) @ emb([q]).T
fused = rrf([rank(bm25.get_scores(tokenize(q))), rank(dense_scores)], k=60)
\end{lstlisting}
\end{bcode}
\begin{bres}[Retrieving the correct answer for 790 TruthfulQA questions]
\begin{tabular}{lrrrrr}
\toprule
Retriever & R@1 & R@5 & MRR & nDCG & Misconception@1\\
\midrule
BM25 & 0.377 & 0.772 & 0.567 & 0.625 & 0.471\\
Dense & 0.384 & 0.737 & 0.547 & 0.600 & 0.381\\
Hybrid (RRF) & \textbf{0.462} & \textbf{0.772} & \textbf{0.609} & \textbf{0.657} & \textbf{0.362}\\
\bottomrule
\end{tabular}
\end{bres}
\begin{bwarn}[Retrieval can return confident falsehoods]
TruthfulQA includes plausible \emph{wrong} answers. BM25 puts a misconception first for 47\% of questions, because wrong answers repeat the question's words. A RAG system is only as truthful as what it retrieves.
\end{bwarn}
\begin{bdef}[Re-ranking]
A second, slower model (a cross-encoder) rescores the top candidates by reading query and passage together.
\end{bdef}
\begin{bres}
On a held-out half, our re-ranker \emph{lowered} R@1 from 0.468 to 0.423. Re-rankers must be validated on your data --- they are not a free win.
\end{bres}
\figw[0.75]{s4_retrieval.png}{Recall@k for BM25, dense and hybrid retrieval on TruthfulQA.}
\begin{bapp}
Web and enterprise search (Elasticsearch, OpenSearch, Vespa) combine BM25 with embeddings; Perplexity and Bing chat run retrieval before every answer.
\end{bapp}

\hourhead{18}{BUILD}{PolicyGPT: chunking and indexing a real PDF}{The AlgoProfessor syllabus PDF (20 pages)}{PDF parsing, chunking, metadata}

\topic{Chunking}
\begin{bdef}[Chunk]
A passage small enough to embed and to fit in a prompt, but large enough to make sense on its own. Each chunk keeps metadata: source file, page, section.
\end{bdef}
\begin{bmath}[Sliding window]
With chunk size $S$ words and overlap $O$, the number of chunks is about $\lceil (N-O)/(S-O)\rceil$ for $N$ words.
\end{bmath}
\begin{balgo}[Ingestion]
\begin{algorithmic}[1]
\For{each page of each PDF}
  \State extract text; normalise whitespace and hyphenation
  \State split into overlapping windows of $S$ words
  \State store (text, file, page) for each chunk
\EndFor
\State embed every chunk and build BM25 + vector indexes
\end{algorithmic}
\end{balgo}
\begin{bcode}
\begin{lstlisting}
for page_no, page in enumerate(PdfReader(path).pages, 1):
    words = clean(page.extract_text()).split()
    for i in range(0, len(words), S - O):
        chunks.append({"text": " ".join(words[i:i+S]), "page": page_no, "file": path.name})
\end{lstlisting}
\end{bcode}
\begin{bres}
20 pages, 4,761 words $\rightarrow$ 79 chunks with page numbers.
\end{bres}
\begin{bwarn}[Chunk boundaries matter]
Splitting a fee table across two chunks means neither chunk answers the fee question. Keep tables and headings with their content; overlap windows.
\end{bwarn}

\hourhead{19}{BUILD}{Grounded answers, citations and abstention}{Syllabus index + golden questions}{Prompting, citations, abstention thresholds}

\topic{Retrieval-augmented generation}
\begin{bdef}[RAG]
Retrieve the passages most relevant to the question, place them in the prompt, and instruct the model to answer \emph{only} from them, citing which passage supports each claim.
\end{bdef}
\begin{bdef}[Abstention]
Answering ``I don't know --- this is not in the documents'' when retrieval finds nothing strong enough. An honest abstention beats a fluent guess.
\end{bdef}
\begin{bmath}[Coverage-based abstention]
coverage$(q)=\dfrac{|\text{content words of }q\text{ found in the best chunk}|}{|\text{content words of }q|}$; \;answer only if coverage $\ge\tau$.
\end{bmath}
\begin{balgo}[PolicyGPT answer]
\begin{algorithmic}[1]
\State retrieve top-$k$ chunks with hybrid search
\If{coverage $<\tau$} \State \Return ``Not found in the documents.'' \EndIf
\State build a prompt: rules + numbered chunks + question
\State generate; keep sentences that cite a chunk [$n$]
\State \Return answer + citations (file, page)
\end{algorithmic}
\end{balgo}
\begin{bcode}[Prompt template]
\begin{lstlisting}
Answer ONLY from the numbered sources. Cite like [2].
If the sources do not contain the answer, reply exactly: NOT_FOUND.
[1] (syllabus.pdf p.3) ... [2] (syllabus.pdf p.17) ...
Question: What is the course fee in India?
\end{lstlisting}
\end{bcode}
\begin{bres}
$\tau=0.50$ was chosen on a development set, never on the test questions. ``What is the hostel fee?'' (not in the syllabus) correctly abstains with coverage 0.40.
\end{bres}
\begin{bwarn}[Our first version abstained too often]
Version 1 wrongly refused 55\% of answerable questions: words like ``what'' and ``fees'' versus ``fee'' were not matching. Light stemming, a question-word stop-list and tuning $\tau$ on the dev set fixed it.
\end{bwarn}
\begin{bapp}
Customer-support copilots, HR and policy assistants, legal research tools and documentation chatbots are built this way; citations let users check every claim.
\end{bapp}

\hourhead{20}{BUILD}{Evaluating RAG}{Golden set --- 20 questions with gold facts}{Retrieval vs generation metrics, failure analysis}

\topic{Two halves, two sets of metrics}
\begin{bdef}[Golden set]
A small, carefully written set of questions with the correct answer, the gold fact and its source page --- including questions that \emph{should} be refused.
\end{bdef}
\begin{bmath}[RAG metrics]
hit@$k$: gold page retrieved in top $k$ \quad answer-correct: answer contains the gold fact\\
faithfulness: fraction of answer claims supported by the cited chunks\\
abstention: correct refusals on unanswerable questions; wrong refusals on answerable ones
\end{bmath}
\begin{balgo}[Failure analysis]
\begin{algorithmic}[1]
\For{each wrong answer}
  \If{gold page not retrieved} \State label \emph{retrieval failure}
  \ElsIf{answer contradicts sources} \State label \emph{faithfulness failure}
  \Else \State label \emph{incomplete answer}
  \EndIf
\EndFor
\State fix the largest bucket first
\end{algorithmic}
\end{balgo}
\begin{bres}[PolicyGPT on the golden set]
\begin{tabular}{lr}
\toprule
Metric & Value\\
\midrule
hit@3 (retrieval) & 0.95\\
Answer contains the gold fact & 0.60\\
Faithfulness & 1.00\\
Wrongly abstained & 0.00\\
Correctly abstained (unanswerable) & 1.00\\
\bottomrule
\end{tabular}\\[3pt]
Retrieval is strong; the gap is in the answer step --- extractive answers sometimes cut off the gold fact.
\end{bres}
\begin{bapp}
Frameworks such as RAGAS, TruLens and DeepEval automate these metrics; production teams run golden sets in CI before every change to the index or prompt.
\end{bapp}

\section*{Check yourself}
\begin{enumerate}
\item Why does HNSW beat IVF on recall here at similar speed? \emph{The graph search adapts to the query; IVF misses neighbours that fall in unprobed clusters.}
\item Why use RRF rather than adding BM25 and cosine scores? \emph{The scores are on different scales; ranks are comparable.}
\item Where should $\tau$ be tuned? \emph{On a development set, never on the test questions.}
\item hit@3 is 0.95 but answer-correct is 0.60. Which component do you fix? \emph{Generation / answer extraction.}
\end{enumerate}
