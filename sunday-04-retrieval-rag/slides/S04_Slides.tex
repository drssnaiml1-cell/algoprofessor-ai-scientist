\begin{frame}{Embeddings and similarity}
\begin{bdef}[Embedding]
A vector representation of text in which texts with similar meaning lie close together. Search then becomes ``find the nearest vectors''.
\end{bdef}
\begin{columns}[T]
\column{0.52\textwidth}
\begin{bmath}[Cosine similarity]
$\cos(u,v)=\dfrac{u\cdot v}{\lVert u\rVert\,\lVert v\rVert}\in[-1,1]$\\[3pt]
For unit vectors: $\cos(u,v)=u^\top v$ (a dot product)
\end{bmath}
\begin{bmath}[LSA embeddings]
TF-IDF matrix $X\approx U_k\Sigma_kV_k^\top$ (truncated SVD); keep $k=256$ dimensions.
\end{bmath}
\column{0.44\textwidth}
\begin{bex}[Worked example]
$u=(0.6,0.8)$, $v=(0.8,0.6)$: $\cos=0.48+0.48=0.96$ --- very similar. $w=(-0.8,0.6)$: $\cos(u,w)=0$ --- unrelated.
\end{bex}
\begin{bdef}[Sentence embeddings]
Neural encoders (e.g.\ \texttt{all-MiniLM-L6-v2}) give better meaning-vectors than LSA; the search code is the same.
\end{bdef}
\end{columns}
\end{frame}

\begin{frame}{Exact and approximate nearest-neighbour search}
\begin{columns}[T]
\column{0.32\textwidth}
\begin{bdef}[Flat (exact)]
Compare the query with every vector: $O(nd)$ per query. Perfect recall, slow at millions.
\end{bdef}
\column{0.32\textwidth}
\begin{balgo}[IVF]
\begin{algorithmic}[1]
\State $k$-means: $n_{list}$ centroids
\State assign vectors to nearest centroid
\State query: pick $n_{probe}$ closest lists
\State search only those lists
\end{algorithmic}
\end{balgo}
\column{0.32\textwidth}
\begin{balgo}[HNSW]
\begin{algorithmic}[1]
\State build layered proximity graph
\State start at top layer entry point
\State greedy walk to closer neighbours
\State descend a layer; repeat
\end{algorithmic}
\end{balgo}
\end{columns}
\begin{bmath}[Recall of approximate search]
$\mathrm{Recall@10}=\frac{|\text{ANN top-10}\cap\text{exact top-10}|}{10}$
\end{bmath}
\end{frame}

\begin{frame}[fragile]{FAISS in code, result and products}
\begin{columns}[T]
\column{0.5\textwidth}
\begin{bcode}
\begin{lstlisting}

\begin{frame}{BM25: the keyword workhorse}
\begin{bdef}[BM25]
A ranking function that rewards query words appearing in a document, with diminishing returns for repeats, more weight for rare words, and a penalty for very long documents.
\end{bdef}
\begin{columns}[T]
\column{0.6\textwidth}
\begin{bmath}[Formula]
$\mathrm{BM25}(q,d)=\sum_{t\in q}\mathrm{IDF}(t)\,\dfrac{f(t,d)\,(k_1+1)}{f(t,d)+k_1\big(1-b+b\,\frac{|d|}{\overline{|d|}}\big)}$\\[3pt]
$\mathrm{IDF}(t)=\ln\!\Big(\frac{N-n_t+0.5}{n_t+0.5}+1\Big)$, \; typically $k_1\approx1.2$--$2$, $b=0.75$
\end{bmath}
\column{0.36\textwidth}
\begin{bex}[Intuition]
``watermelon seeds'': \emph{watermelon} is rare $\Rightarrow$ high IDF; \emph{seeds} common $\Rightarrow$ lower. A 10th repeat of a word adds far less than the 1st.
\end{bex}
\end{columns}
\end{frame}

\begin{frame}{Hybrid fusion and retrieval metrics}
\begin{columns}[T]
\column{0.5\textwidth}
\begin{bdef}[Hybrid retrieval]
Combine keyword (BM25) and dense (embedding) rankings: keywords catch exact names and numbers, embeddings catch paraphrases.
\end{bdef}
\begin{bmath}[Reciprocal-rank fusion]
$\mathrm{RRF}(d)=\sum_{r\in\text{retrievers}}\dfrac{1}{k+\mathrm{rank}_r(d)},\quad k=60$
\end{bmath}
\begin{bex}[RRF by hand]
Doc ranked 1st by BM25 and 3rd by dense: $\frac1{61}+\frac1{63}=0.0323$.
\end{bex}
\column{0.46\textwidth}
\begin{bmath}[Metrics]
Recall@$k$ $=$ share of queries with the gold doc in the top $k$\\[2pt]
$\mathrm{MRR}=\frac{1}{|Q|}\sum_q\frac{1}{\mathrm{rank}_q}$\\[2pt]
$\mathrm{nDCG@10}=\frac{1}{\log_2(\mathrm{rank}+1)}$ if rank $\le 10$, else 0 (one gold doc)
\end{bmath}
\begin{bdef}[Re-ranker]
A second, slower model that rescores the top candidates. A \emph{cross-encoder} reads query and passage together.
\end{bdef}
\end{columns}
\end{frame}

\begin{frame}{Retrieval-augmented generation (RAG)}
\begin{bdef}[RAG]
Retrieve relevant passages from your own documents and give them to the generator, so answers are grounded in sources you control --- and can cite them.
\end{bdef}
\begin{center}
\begin{tikzpicture}[every node/.style={font=\small,rounded corners,minimum height=0.85cm,minimum width=2.0cm,align=center}]
\node[fill=navy,text=gold] (q) at (0,0) {Question};
\node[fill=sun,text=white] (r) at (2.7,0) {Retrieve\\top-$k$ chunks};
\node[fill=gold,text=navy] (c) at (5.4,0) {Coverage\\check};
\node[fill=dataC,text=white] (g) at (8.1,0) {Generate /\\extract answer};
\node[fill=saffron,text=white] (a) at (10.8,0) {Answer +\\citations};
\node[fill=warnC,text=white] (x) at (5.4,-1.5) {``I can't find this''};
\foreach \u/\v in {q/r,r/c,c/g,g/a} \draw[-{Stealth},thick,navy] (\u)--(\v);
\draw[-{Stealth},thick,warnC] (c)--(x) node[midway,right,font=\scriptsize]{low};
\end{tikzpicture}
\end{center}
\begin{bdef}[Ingestion]
Offline: parse documents $\rightarrow$ clean $\rightarrow$ chunk $\rightarrow$ embed / index, keeping metadata (file, page, section) on every chunk.
\end{bdef}
\end{frame}

\begin{frame}[fragile]{Chunking: algorithm, maths, code and result}
\begin{columns}[T]
\column{0.48\textwidth}
\begin{balgo}[Sliding-window chunking]
\begin{algorithmic}[1]
\For{each page $p$}
  \State words $w\gets$ clean text of $p$
  \For{$s=0,\,L{-}o,\,2(L{-}o),\dots$}
    \State chunk $\gets w[s:s+L]$, tag page $p$
  \EndFor
\EndFor
\end{algorithmic}
\end{balgo}
\begin{bmath}[How many chunks?]
$\approx\left\lceil\frac{W-o}{L-o}\right\rceil$ per page ($L$ size, $o$ overlap)
\end{bmath}
\column{0.48\textwidth}
\begin{bcode}
\begin{lstlisting}
reader = PdfReader(pdf_path)
pages = [clean(p.extract_text())
         for p in reader.pages]
chunks = chunk_pages(pages, 90, 30)
\end{lstlisting}
\end{bcode}
\begin{bres}
20 pages, 4,761 words $\rightarrow$ \textbf{79 chunks} of about 90 words, each with its page number.
\end{bres}
\begin{bex}[Trade-off]
Small chunks: precise citations, less context. Large: more context, diluted relevance.
\end{bex}
\end{columns}
\end{frame}

\begin{frame}{Grounding, citations and abstention}
\begin{columns}[T]
\column{0.5\textwidth}
\begin{bdef}[Grounded answer]
Every claim in the answer is supported by a retrieved passage, and the passage is cited.
\end{bdef}
\begin{bdef}[Abstention]
Refusing to answer when the documents do not contain the answer --- the main defence against hallucination.
\end{bdef}
\begin{bmath}[Coverage]
$\mathrm{cov}(q)=\dfrac{\sum_{w\in q\cap\text{chunks}}\mathrm{idf}(w)}{\sum_{w\in q}\mathrm{idf}(w)}$; \; abstain if $\mathrm{cov}<\tau$
\end{bmath}
\column{0.46\textwidth}
\begin{balgo}[Answer pipeline]
\begin{algorithmic}[1]
\State hybrid-retrieve top 3 chunks
\If{$\mathrm{cov}(q)<\tau$}
  \State \Return ``I can't find this''
\EndIf
\State pick best 25-word span
\State \Return span + page citations
\end{algorithmic}
\end{balgo}
\end{columns}
\end{frame}

\hourslide{20}{BUILD}{Evaluate: dev set to tune, golden set to test}{8 dev + 25 golden questions}{RAG evaluation, hallucination measurement, regression tests}

\begin{frame}{Evaluation design}
\begin{columns}[T]
\column{0.5\textwidth}
\begin{bdef}[Dev set vs golden set]
\textbf{Dev}: used to choose settings (threshold $\tau$). \textbf{Golden}: never touched until the final report. Reporting on the tuning set overstates quality.
\end{bdef}
\begin{bmath}[Metrics]
hit@3 $=$ gold fact in top-3 chunks\\
answer accuracy $=$ gold fact in the answer\\
abstention accuracy $=$ share of unanswerable questions refused\\
faithfulness $=$ share of answer words found in cited chunks
\end{bmath}
\column{0.46\textwidth}
\begin{balgo}[Tune the threshold]
\begin{algorithmic}[1]
\For{$\tau\in\{0.30,0.35,\dots,0.90\}$}
  \State score$(\tau)$ on the dev set
\EndFor
\State $\tau^\ast\gets$ middle of the best plateau
\State report once on the golden set
\end{algorithmic}
\end{balgo}
\begin{bapp}
RAG evaluation tools: RAGAS, TruLens, LangSmith; keep the golden set as a regression test in CI.
\end{bapp}
\end{columns}
\end{frame}
