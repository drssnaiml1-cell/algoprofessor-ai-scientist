Small chunks: precise citations, less context.\\ Large chunks: more context, diluted relevance.\\ Overlap: facts are not cut in half.

\hourslide{19}{BUILD}{Cited Q\&A API with abstention}{Course syllabus PDF}{Hybrid search, citations, abstention, FastAPI}

\begin{frame}{Knowing when to say ``I don't know''}
\[\mathrm{coverage}(q)=\frac{\sum_{w\in q\cap\text{chunks}}\mathrm{idf}(w)}{\sum_{w\in q}\mathrm{idf}(w)}\quad\text{abstain if below threshold}\]
\begin{columns}[T]
\column{0.5\textwidth}
\begin{exampleblock}{Answered, with citation}
``What is the programme fee in India?''\\ $\rightarrow$ ``\ldots INR 50,000 (Hyderabad \& Shirdi)'' [p.\,1]
\end{exampleblock}
\column{0.46\textwidth}
\begin{alertblock}{Abstained}
``What is the hostel fee?'' --- ``fee'' matches, ``hostel'' never appears. Coverage 0.40.
\end{alertblock}
\end{columns}
\end{frame}
