\begin{frame}{What MLOps is}
\begin{bdef}[MLOps]
The practices that take a model from a notebook to a reliable service and keep it reliable: versioning, testing, packaging, automated deployment and monitoring.
\end{bdef}

\begin{columns}[T]
\column{0.32\textwidth}
\begin{bdef}[REST API]
A URL endpoint that takes JSON in and returns JSON out.
\end{bdef}
\column{0.32\textwidth}
\begin{bdef}[Container]
Code + dependencies packaged to run the same everywhere.
\end{bdef}
\column{0.32\textwidth}
\begin{bdef}[CI/CD]
Every push runs tests and builds automatically; green builds deploy.
\end{bdef}

\begin{frame}{CI/CD and the model registry}
\begin{columns}[T]
\column{0.5\textwidth}
\begin{balgo}[CI pipeline on every push]
\begin{algorithmic}[1]
\State check out code
\State install pinned dependencies
\State run unit and API tests
\If{any test fails} \State stop; nothing deploys \EndIf
\State build container image tagged with commit
\State deploy to staging, then production
\end{algorithmic}
\end{balgo}
\column{0.46\textwidth}
\begin{bdef}[Model registry]
Stores every model version with its metrics, data version and stage (staging, production, archived), so you can compare and roll back.
\end{bdef}
\begin{bapp}
GitHub Actions, GitLab CI; MLflow Model Registry; managed platforms such as AWS SageMaker, Google Vertex AI, Azure ML; Kubernetes for scaling.
\end{bapp}
\end{columns}
\end{frame}

\begin{frame}{Drift: is today's data still like training data?}
\begin{columns}[T]
\column{0.5\textwidth}
\begin{bdef}[Data drift vs concept drift]
\textbf{Data drift}: inputs change, $p(x)$ shifts. \textbf{Concept drift}: the relationship changes, $p(y\mid x)$ shifts.
\end{bdef}
\begin{bmath}[PSI and KS]
$\mathrm{PSI}=\sum_i (c_i-r_i)\ln\frac{c_i}{r_i}$ \;(bins of reference)\\
$<0.1$ stable, $0.1$--$0.25$ moderate, $>0.25$ major\\[2pt]
$D_{KS}=\sup_x|F_{ref}(x)-F_{cur}(x)|$
\end{bmath}
\begin{bex}[PSI by hand]
Bin share $r=0.5\rightarrow c=0.3$: $(0.3-0.5)\ln(0.6)=0.102$.
\end{bex}
\column{0.46\textwidth}
\begin{bres}
Coastal $\rightarrow$ inland: PSI longitude 2.03, latitude 1.07, income 0.33. RMSE \$52,815 $\rightarrow$ \alert{\textbf{\$76,285}}.
\end{bres}
\centering\includegraphics[width=\linewidth]{s6_drift.png}
\begin{bapp}
Evidently, WhyLabs, Arize; cloud model monitors.
\end{bapp}
\end{columns}
\end{frame}

\begin{frame}{Fairness: definitions and metrics}
\begin{bdef}[Algorithmic fairness]
A model is unfair when its errors or benefits differ systematically across groups (e.g.\ by sex or caste) in ways that cannot be justified for the decision.
\end{bdef}
\begin{columns}[T]
\column{0.55\textwidth}
\begin{bmath}[Two common metrics]
Demographic parity difference:\\ $\;|P(\hat y=1\mid A=a)-P(\hat y=1\mid A=b)|$\\[2pt]
Equalized odds difference:\\ $\;\max\big(|\mathrm{TPR}_a-\mathrm{TPR}_b|,\;|\mathrm{FPR}_a-\mathrm{FPR}_b|\big)$
\end{bmath}
\column{0.41\textwidth}
\begin{bex}[Our baseline]
Selection rate women 0.073, men 0.262 $\Rightarrow$ DP difference 0.189. TPR 0.494 vs 0.631; FPR 0.020 vs 0.101.
\end{bex}
\begin{bdef}[Proxy discrimination]
\texttt{sex} was not an input; marital status and relationship carried it in.
\end{bdef}
\end{columns}
\end{frame}

\begin{frame}[fragile]{Mitigation: algorithm, maths, code, result}
\begin{columns}[T]
\column{0.5\textwidth}
\begin{bmath}[Reweighing (pre-processing)]
$w(g,y)=\dfrac{P(G=g)\,P(Y=y)}{P(G=g,\,Y=y)}$ --- makes group and label look independent
\end{bmath}
\begin{balgo}[Fairness audit]
\begin{algorithmic}[1]
\State choose groups and a fairness definition
\State measure metrics per group
\State try pre- / in- / post-processing fixes
\State report accuracy--fairness trade-off
\State a human decides; document why
\end{algorithmic}
\end{balgo}
\column{0.46\textwidth}
\begin{bcode}
\begin{lstlisting}
to = ThresholdOptimizer(estimator=base,
       constraints="equalized_odds", prefit=True)
to.fit(X_tr, y_tr, sensitive_features=s_tr)
\end{lstlisting}
\end{bcode}
\begin{bres}
\begin{tabular}{lrrr}
 & Acc & DP & EO\\
Baseline & 0.854 & 0.189 & 0.137\\
Reweighing & 0.849 & 0.102 & 0.082\\
EO thresholds & 0.838 & 0.097 & \textbf{0.055}\\
\end{tabular}
\end{bres}
\end{columns}
\end{frame}

\begin{frame}{Red-teaming and AI law}
\begin{columns}[T]
\column{0.5\textwidth}
\begin{bdef}[Red-teaming]
Deliberately attacking your own system to find failures before users do.
\end{bdef}
\begin{balgo}[Red-team cycle]
\begin{algorithmic}[1]
\For{each risk category}
  \State write probes; run them
  \State log output, severity, fix
\EndFor
\State re-test after every fix
\end{algorithmic}
\end{balgo}
\begin{bres}
Naive retrieval repeats the misconception \textbf{47.1\%} of the time.
\end{bres}
\column{0.46\textwidth}
\begin{bdef}[EU AI Act]
Risk tiers: \emph{unacceptable} (banned) $\cdot$ \emph{high-risk} (hiring, credit, education: risk management, human oversight, logging) $\cdot$ \emph{limited} (transparency) $\cdot$ \emph{minimal}.
\end{bdef}
\begin{bdef}[India DPDP Act 2023]
Purpose and consent $\cdot$ rights to access, correct, erase $\cdot$ breach reporting $\cdot$ Data Protection Board.
\end{bdef}
\begin{bapp}
Fairlearn, IBM AIF360, MS RAI Toolbox.
\end{bapp}
\end{columns}
\end{frame}

\begin{frame}{Readiness: definition, checklist, result}
\begin{columns}[T]
\column{0.5\textwidth}
\begin{bdef}[Production readiness]
Evidence that a system can be run, changed and trusted by people other than its author.
\end{bdef}
\begin{balgo}[Automated audit]
\begin{algorithmic}[1]
\For{each check $c$ in the checklist}
  \State pass$_c\gets c(\text{repo})$
\EndFor
\State score $\gets\sum_c$ pass$_c$
\State open a ticket for each failed check
\end{algorithmic}
\end{balgo}
\column{0.46\textwidth}
\begin{bex}[Ten checks]
README sections $\cdot$ requirements $\cdot$ tests exist $\cdot$ tests pass $\cdot$ Dockerfile $\cdot$ CI $\cdot$ model card $\cdot$ fairness/safety audit $\cdot$ LICENSE $\cdot$ no secrets
\end{bex}
\begin{bres}
Demo capstone \textbf{8/10}: missing \texttt{docs/audit.md} and \texttt{LICENSE}.
\end{bres}
\begin{bapp}
Industry rubrics such as Google's ``ML Test Score'' for production ML systems.
\end{bapp}
\end{columns}
\end{frame}
