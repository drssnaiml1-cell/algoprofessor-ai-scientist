\hourslide{22}{TEACH}{Planner, executor, critic}{GSM8K test set (1,319 problems)}{LangGraph-style state machines, self-critique, guardrails}

\begin{frame}{An agent team that checks reasoning}
\begin{columns}[T]
\column{0.5\textwidth}
\begin{enumerate}
\item \textbf{Planner:} extract \texttt{<<16-3-4=9>>} steps
\item \textbf{Executor:} recompute with the calculator
\item \textbf{Critic:} accept or flag
\end{enumerate}
\begin{exampleblock}{Our run}
1,301 accepted $\cdot$ 18 no steps $\cdot$ \textbf{0 flagged}\\ 3.25 steps per problem
\end{exampleblock}
\column{0.46\textwidth}
\begin{alertblock}{First run flagged 6}
All were format noise (\texttt{+8}, \texttt{3/4}), not maths errors. Inspect failures before trusting a rate.
\end{alertblock}
\begin{block}{Guardrails}
Arithmetic-only tool $\cdot$ step limit $\cdot$ hop limit $\cdot$ memory for audit
\end{block}
\end{columns}
\end{frame}

\hourslide{23}{BUILD}{ScholarAgent: an AI research-scientist agent}{arXiv API (live in class) / offline index}{API tools, rate limits, citation tracking}

\begin{frame}{Cite only what you retrieved}
\begin{columns}[T]
\column{0.5\textwidth}
\begin{enumerate}
\item Search arXiv; on failure use the offline index \emph{and say so}
\item Keep clearly relevant papers only
\item Write a numbered, cited summary
\end{enumerate}
\column{0.46\textwidth}
\begin{exampleblock}{Our run}
\small ``efficient fine-tuning \ldots low rank adaptation'' $\rightarrow$ \emph{LoRA}, Hu et al.\ (2021), arXiv:2106.09685
\end{exampleblock}
\begin{alertblock}{Never invent a citation}
Stop rather than guess when nothing is found.
\end{alertblock}
\end{columns}
\end{frame}

\hourslide{24}{BUILD}{Sandboxed code-execution agent}{UCI Adult (baseline reproduction)}{Sandboxing, permission boundaries, reproducibility}

\begin{frame}{Defence in depth}
\begin{columns}[T]
\column{0.5\textwidth}
\begin{enumerate}
\item Static check: no OS, network or \texttt{eval}
\item Separate isolated process
\item Memory cap, time limit, empty environment
\item Production: container with \texttt{--network none}
\end{enumerate}
\column{0.46\textwidth}
\begin{exampleblock}{Our run}
Baseline on Adult: \textbf{0.8515 $\pm$ 0.0023} (5-fold)
\end{exampleblock}
\begin{alertblock}{Blocked}
\texttt{import os} $\cdot$ \texttt{\_\_import\_\_} $\cdot$ a 10-billion-item list (memory limit)
\end{alertblock}
\end{columns}
\end{frame}
