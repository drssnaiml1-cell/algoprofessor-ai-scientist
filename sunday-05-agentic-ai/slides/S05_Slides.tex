\begin{frame}{What is an AI agent?}
\begin{bdef}[Agent]
A system in which a model decides, step by step, which action to take --- call a tool, read the result, decide again --- until it reaches a goal or a limit. \textbf{Agent = model + tools + loop + memory + guardrails.}
\end{bdef}
\begin{columns}[T]
\column{0.52\textwidth}
\begin{bmath}[The agent as a decision loop]
State $s_t=(\text{goal},\,\text{history }h_{<t})$\\
Action $a_t\sim\pi(a\mid s_t)$, \; $a_t\in\text{Tools}\cup\{\text{answer}\}$\\
Observation $o_t=\text{tool}(a_t)$, \; $s_{t+1}=s_t\oplus(a_t,o_t)$\\
Stop when $a_t=\text{answer}$ or $t=T_{max}$
\end{bmath}
\column{0.44\textwidth}
\begin{bex}[Chatbot vs agent]
A chatbot \emph{answers} ``What is the fee in dollars?'' from memory. An agent \emph{searches} the fee, \emph{calls} a calculator, then answers --- with evidence.
\end{bex}
\end{columns}
\end{frame}

\begin{frame}{Function calling and the ReAct loop}
\begin{columns}[T]
\column{0.5\textwidth}
\begin{bdef}[Function calling]
The model outputs a structured call, e.g.\ \texttt{\{"tool": "lookup\_hour", "args": \{"hour": 6\}\}}. Your code validates and runs it, then returns the result. The model never executes anything itself.
\end{bdef}
\begin{bdef}[Tool schema]
Name, description, and a JSON Schema for the arguments --- the model reads these to decide what to call.
\end{bdef}
\column{0.46\textwidth}
\begin{balgo}[ReAct]
\begin{algorithmic}[1]
\Require question $q$, tools, $T_{max}$
\State history $\gets[\,]$
\For{$t=1$ to $T_{max}$}
  \State thought, action $\gets$ model$(q$, history$)$
  \If{action is ``answer''} \State \Return answer \EndIf
  \State obs $\gets$ run tool(action)
  \State append (thought, action, obs)
\EndFor
\State \Return ``stopped: step limit''
\end{algorithmic}
\end{balgo}
\end{columns}
\end{frame}

\begin{frame}{Multi-agent graphs}
\begin{columns}[T]
\column{0.5\textwidth}
\begin{bdef}[Agent graph]
Nodes are agents or steps; edges decide what runs next based on a shared state. Gives control, loops with limits, and an audit trail.
\end{bdef}
\begin{bmath}[State machine]
$G=(N,\,E,\,s_0)$, \; node $n:\;s\mapsto s'$\\
router $r_n:\;s'\mapsto$ next node or END\\
run: $s_{k+1}=n_k(s_k)$, \; $n_{k+1}=r_{n_k}(s_{k+1})$
\end{bmath}
\begin{bdef}[Memory]
\textbf{Short-term}: the current state. \textbf{Long-term}: stored results reused later (here: every verdict).
\end{bdef}
\column{0.46\textwidth}
\begin{center}
\begin{tikzpicture}[every node/.style={font=\small,rounded corners,minimum width=2.4cm,minimum height=0.8cm,align=center}]
\node[fill=navy,text=gold] (p) at (0,0) {Planner\\extract steps};
\node[fill=sun,text=white] (e) at (0,-1.4) {Executor\\recompute (tool)};
\node[fill=gold,text=navy] (c) at (0,-2.8) {Critic\\accept / flag};
\node[fill=gray!40] (m) at (2.9,-1.4) {Memory};
\draw[-{Stealth},thick] (p)--(e); \draw[-{Stealth},thick] (e)--(c);
\draw[-{Stealth},thick,dashed] (c.east) -| (m.south);
\end{tikzpicture}
\end{center}
\begin{bex}[GSM8K trace]
``16 $-$ 3 $-$ 4 = \texttt{<<16-3-4=9>>}9 eggs'' --- the executor recomputes $16-3-4$ and the critic compares with 9.
\end{bex}
\end{columns}
\end{frame}

\begin{frame}{Working with external APIs}
\begin{columns}[T]
\column{0.5\textwidth}
\begin{bdef}[Rate limit]
A cap on requests per time window set by the API provider. Exceeding it returns errors (often HTTP 429).
\end{bdef}
\begin{bmath}[Token bucket and backoff]
Bucket capacity $C$, refill $r$ tokens/s: a request may run if tokens $\ge1$.\\[2pt]
Retry wait $=b\cdot2^{k}$ (+ jitter) after the $k$-th failure: $1, 2, 4, 8$\,s \ldots
\end{bmath}
\begin{bdef}[Citation tracking]
Every claim in the output carries the id of the source it came from.
\end{bdef}
\column{0.46\textwidth}
\begin{balgo}[ScholarAgent]
\begin{algorithmic}[1]
\Require topic $q$
\State try arXiv API (with retries, timeout)
\If{network fails}
  \State use offline index; \textbf{say so}
\EndIf
\State keep papers with score $\ge\frac12$ best
\If{none left} \State \Return ``no sources'' \EndIf
\State write numbered summary, cite $[n]$
\end{algorithmic}
\end{balgo}
\end{columns}
\end{frame}

\begin{frame}{Sandboxing model-written code}
\begin{columns}[T]
\column{0.5\textwidth}
\begin{bdef}[Sandbox]
An isolated environment where untrusted code runs with limited permissions, resources and network access, so mistakes or attacks cannot harm the host.
\end{bdef}
\begin{bdef}[Threat model]
Delete files $\cdot$ read secrets $\cdot$ exfiltrate data over the network $\cdot$ exhaust memory or CPU $\cdot$ run forever.
\end{bdef}
\begin{bmath}[Resource limits]
memory $\le$ 2\,GB $\cdot$ wall-time $\le$ 60\,s $\cdot$ env $=\varnothing$ $\cdot$ network = none (container)
\end{bmath}
\column{0.46\textwidth}
\begin{balgo}[Defence in depth]
\begin{algorithmic}[1]
\State static check: no \texttt{os}, \texttt{socket}, \texttt{eval}
\If{check fails} \State \Return refused \EndIf
\State write code to a temp directory
\State run isolated process with limits
\State capture stdout/stderr, enforce timeout
\State delete the temp directory
\end{algorithmic}
\end{balgo}
\end{columns}
\end{frame}

\begin{frame}{How to evaluate an agent}
\begin{columns}[T]
\column{0.5\textwidth}
\begin{bdef}[Agent evaluation]
Run a fixed suite of tasks; record success, steps, tool calls, latency and cost; classify every failure.
\end{bdef}
\begin{bmath}[Success rate with uncertainty]
$\hat p=\frac{\text{successes}}{n}$, \; $\mathrm{SE}=\sqrt{\hat p(1-\hat p)/n}$\\
$7/12$: $\hat p=0.583\pm0.142$ --- small suites are noisy\\[2pt]
cost/task $=\frac{\text{tokens}}{1000}\times$ price per 1k tokens
\end{bmath}
\column{0.46\textwidth}
\begin{balgo}[Evaluation harness]
\begin{algorithmic}[1]
\For{each task $(q, \text{gold})$ in suite}
  \State run agent with tracing on
  \State record steps, tools, ms, tokens
  \State classify the outcome
\EndFor
\State report rates, cost, taxonomy
\end{algorithmic}
\end{balgo}
\begin{bdef}[Tracing and human-in-the-loop]
Log every step as a \emph{span} (tool, args, latency, tokens). Irreversible actions (send, pay, delete) wait for a person's approval.
\end{bdef}
\end{columns}
\end{frame}
